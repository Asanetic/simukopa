report Loans "Loans Performance"
  nest loans
  by month|day|year
  date date_issued
  entity device_loans_performance
  role view_reports
  currency KES
  cache 60
  maxrange 1200

  filter payment_frequency = payment_frequency
  filter duration_type = duration_type

  loans = count loans "Loans Issued"
  principal = sum loans.principal "Principal Lent" money
  interest = sum loans.interest "Interest Charged" money
  deposits = sum loans.deposit "Deposits Received" money
  expected = principal + interest "Expected Repayment" money
  paid = sum loans.total_paid "Total Repaid" money
  balance = sum loans.balance "Outstanding Balance" money
  avg_loan = principal / loans "Average Loan Size" money
  chg = change(principal) "Lending vs Previous Period"

  forecast loans next 10
  forecast principal next 10

  headline {label}: {loans} loans issued, {principal} lent, {balance} outstanding.

  remark loans == 0 | watch | No loans | No loans were issued in {label}.
  remark balance > principal | bad | High outstanding balance | Outstanding balance of {balance} exceeds principal lent of {principal}.
  remark chg !== null && chg <= -20 | watch | Lending declined | Principal lent fell by {abs(chg)}% compared with {prev_label}.
  remark true | neutral | Loan performance | {loans} loans issued, {paid} repaid and {balance} outstanding.


report Allocations "Device Allocations"
  nest devices
  by month|day|year
  date allocation_date
  entity device_allocations_performance
  role view_reports
  currency KES
  cache 60
  maxrange 1200

  filter status = status

  allocated = count devices_allocations "Devices Allocated"
  active = count devices_allocations where status=active "Active Allocations"
  returned = count devices_allocations where status=returned "Returned Devices"
  chg = change(allocated) "Allocations vs Previous Period"

  forecast allocated next 10

  headline {label}: {allocated} devices allocated, {active} active and {returned} returned.

  remark allocated == 0 | watch | No allocations | No devices were allocated in {label}.
  remark returned > 0 | watch | Device returns | {returned} devices were returned in {label}.
  remark chg !== null && chg <= -20 | watch | Allocations declined | Device allocations fell by {abs(chg)}% compared with {prev_label}.
  remark true | neutral | Allocation activity | {allocated} devices allocated, including {active} active allocations.

report ClientRepayments "Client Repayment Analysis"
  nest clients
  by dimension
  date date_registered
  entity device_client_repayment_analysis
  role view_reports
  currency KES
  cache 60
  maxrange 1200

  group client_id
  rows "Client"
  labels clients.client_id > name
  sort balance desc

  loans_count = count loans "Loans Taken"
  principal = sum loans.principal "Principal Borrowed" money
  paid = sum loans.total_paid "Total Repaid" money
  balance = sum loans.balance "Outstanding Balance" money
  payments = count loans_repayments "Repayment Transactions"
  collected = sum loans_repayments.paid_amount "Payments Made" money
  avg_payment = collected / payments "Average Payment" money

  headline {label}: {loans_count} loans, {principal} borrowed and {balance} outstanding.

  remark loans_count == 0 | neutral | No loans | {label} has no loans recorded.
  remark balance > 0 && paid == 0 | bad | No repayments | {label} has an outstanding balance of {balance} with no recorded repayments.
  remark balance > 0 && paid > 0 | watch | Balance remaining | {label} has repaid {paid} and still owes {balance}.
  remark true | good | Repayment status | {label}: {paid} repaid across {payments} transactions.


report RepaymentByLoan "Repayments by Loan"
  nest repayments
  by dimension
  date payment_date
  entity device_repayments_by_loan
  role view_reports
  currency KES
  cache 60
  maxrange 1200

  group loan_id
  rows "Loan"
  labels loans.loan_id > loan_id
  sort collected desc

  payments = count loans_repayments "Payments"
  collected = sum loans_repayments.paid_amount "Total Collected" money
  remaining = sum loans_repayments.new_balance "Recorded New Balances" money

  headline {label}: {payments} payments received, {collected} collected.

  remark payments == 0 | watch | No repayments | No repayment transactions were recorded for {label}.
  remark remaining > 0 && collected == 0 | bad | No collections | {label} has a recorded balance of {remaining} with no collections.
  remark true | neutral | Loan repayment history | {label}: {collected} collected across {payments} payments.

  cache 60

  loans = count loans "Loans Issued"
